// app\api\contact\route.ts
import { NextResponse } from "next/server";
import { adminDb } from "@/lib/firebase/admin";
import { contactFormSchema } from "@/lib/validations/contact";

export async function POST(request: Request) {
    try {
        const body = await request.json();

        const result = contactFormSchema.safeParse(body);

        if (!result.success) {
            return NextResponse.json(
                {
                    success: false,
                    message: "Invalid form data.",
                    errors: result.error.flatten()
                        .fieldErrors,
                },
                { status: 400 }
            );
        }

        const {
            name,
            email,
            phone,
            brandName,
            message,
            locale,
        } = result.data;

        const docRef = await adminDb
            .collection("contact_submissions")
            .add({
                name,
                email,
                phone,
                brandName: brandName || "",
                message,
                locale,
                status: "new",
                createdAt:
                    new Date(),
            });

        return NextResponse.json(
            {
                success: true,
                id: docRef.id,
            },
            { status: 201 }
        );
    } catch (error) {
        console.error(
            "Contact form submission error:",
            error
        );

        return NextResponse.json(
            {
                success: false,
                message:
                    "Something went wrong while submitting the form.",
            },
            { status: 500 }
        );
    }
}

function encryptTicketToken(token: string): string {

function decryptTicketToken(encryptedToken: string): string {

export async function getBlogPosts({
    category,
    sort = "latest",
    page = 1,
}: {
    category?: string;
    sort?: "latest" | "oldest";
    page?: number;
} = {}): Promise<BlogPostsResult> {
    const requestedPage = Number.isFinite(page)
        ? Math.max(1, Math.floor(page))
        : 1;

    const categories = await getBlogCategories();

    const normalizedCategory = category?.trim();

    const selectedCategory = normalizedCategory
        ? categories.find(
            (item) => item.slug === normalizedCategory
        )
        : undefined;

    if (normalizedCategory && !selectedCategory) {
        return {
            posts: [],
            totalPosts: 0,
            totalPages: 1,
            currentPage: 1,
        };
    }

    const categoryId = selectedCategory?.id ?? null;

    // Run count and posts queries simultaneously.
    const [totalPosts, requestedPosts] = await Promise.all([
        getCachedBlogPostsCount(categoryId),

        getCachedBlogPostsPage(
            categoryId,
            sort,
            requestedPage,
            categories
        ),
    ]);

    const totalPages = Math.max(
        1,
        Math.ceil(totalPosts / POSTS_PER_PAGE)
    );

    const safePage = Math.min(
        requestedPage,
        totalPages
    );

    // Fetch the last valid page if the requested page is out of range.
    const posts =
        safePage === requestedPage
            ? requestedPosts
            : await getCachedBlogPostsPage(
                categoryId,
                sort,
                safePage,
                categories
            );

    return {
        posts,
        totalPosts,
        totalPages,
        currentPage: safePage,
    };
}

function chunkArray<T>(
    items: T[],
    size: number
): T[][] {
    const chunks: T[][] = [];

    for (
        let index = 0;
        index < items.length;
        index += size
    ) {
        chunks.push(items.slice(index, index + size));
    }

    return chunks;
}

/* Cached data layer. */
async function getCachedRelatedBlogPosts(
    categoryIds: string[]
): Promise<BlogPost[]> {
    "use cache";

    cacheLife({
        stale: 60,
        revalidate: 300,
        expire: 3600,
    });

    cacheTag(BLOG_RELATED_POSTS_TAG);

    if (categoryIds.length === 0) {
        return [];
    }

    const categoryChunks = chunkArray(
        categoryIds,
        30
    );

    // Fetch only the latest 30 matching posts per query.
    const snapshots = await Promise.all(
        categoryChunks.map((chunk) =>
            adminDb
                .collection("blog_posts")
                .where("status", "==", "published")
                .where(
                    "categories",
                    "array-contains-any",
                    chunk
                )
                .orderBy("created_at", "desc")
                .limit(RELATED_POSTS_CANDIDATE_LIMIT)
                .get()
        )
    );

    // Deduplicate documents returned by multiple queries.
    const documentsMap = new Map<
        string,
        FirebaseFirestore.QueryDocumentSnapshot
    >();

    for (const snapshot of snapshots) {
        for (const doc of snapshot.docs) {
            documentsMap.set(doc.id, doc);
        }
    }

    if (documentsMap.size === 0) {
        return [];
    }

    const currentCategoryIds = new Set(categoryIds);

    // Rank candidates by shared categories, then creation date.
    const candidates = Array.from(
        documentsMap.values()
    )
        .map((doc) => {
            const data = doc.data();

            const relatedCategoryIds =
                getCategoryIds(data);

            const sharedCategories =
                relatedCategoryIds.filter(
                    (categoryId) =>
                        currentCategoryIds.has(categoryId)
                ).length;

            return {
                doc,
                data,
                sharedCategories,
                createdAt: normalizeDate(data.created_at),
            };
        })
        .filter(
            (candidate) =>
                candidate.sharedCategories > 0
        )
        .sort((a, b) => {
            if (
                a.sharedCategories !==
                b.sharedCategories
            ) {
                return (
                    b.sharedCategories -
                    a.sharedCategories
                );
            }

            return (
                (b.createdAt?.getTime() ?? 0) -
                (a.createdAt?.getTime() ?? 0)
            );
        })
        .slice(0, RELATED_POSTS_CANDIDATE_LIMIT);

    if (candidates.length === 0) {
        return [];
    }

    // Collect only category IDs needed by selected candidates.
    const relatedCategoryIds = [
        ...new Set(
            candidates.flatMap((candidate) =>
                getCategoryIds(candidate.data)
            )
        ),
    ];

    const categoriesMap = new Map<
        string,
        BlogCategory
    >();

    if (relatedCategoryIds.length > 0) {
        const categoryRefs = relatedCategoryIds.map(
            (categoryId) =>
                adminDb
                    .collection("blog_categories")
                    .doc(categoryId)
        );

        const categorySnapshots = await adminDb.getAll(
            ...categoryRefs
        );

        for (const categorySnapshot of categorySnapshots) {
            if (!categorySnapshot.exists) {
                continue;
            }

            categoriesMap.set(
                categorySnapshot.id,
                normalizeCategory(
                    categorySnapshot.id,
                    categorySnapshot.data() ?? {}
                )
            );
        }
    }

    return candidates.map((candidate) =>
        normalizeBlogPost(
            candidate.doc.id,
            candidate.data,
            categoriesMap
        )
    );
}

export async function getRelatedBlogPosts(
    post: BlogPost,
    limit = 3
): Promise<BlogPost[]> {
    const safeLimit = Number.isFinite(limit)
        ? Math.max(0, Math.floor(limit))
        : 3;

    if (
        !post.id ||
        safeLimit === 0 ||
        !post.categories ||
        post.categories.length === 0
    ) {
        return [];
    }

    const categoryIds = [
        ...new Set(
            post.categories
                .map((category) => category.id)
                .filter(
                    (categoryId) =>
                        typeof categoryId === "string" &&
                        categoryId.length > 0
                )
        ),
    ].sort();

    if (categoryIds.length === 0) {
        return [];
    }

    const relatedPosts =
        await getCachedRelatedBlogPosts(categoryIds);

    // Exclude the current article after retrieving cached candidates.
    return relatedPosts
        .filter(
            (relatedPost) =>
                relatedPost.id !== post.id
        )
        .slice(0, safeLimit);
}

function normalizeSession(
    id: string,
    data: SessionDocument
): Session | null {
    const poster =
        data.poster &&
        typeof data.poster === "object"
            ? (data.poster as Record<string, unknown>)
            : null;

        typeof data.youtubeUrl === "string"
            ? data.youtubeUrl.trim()
            : "";

        poster &&
        typeof poster.url === "string"

            : "";

        id,
        title: normalizeLocalizedText(data.title),
        poster: {
            public_id:
                poster &&
                typeof poster.public_id === "string"
                    ? poster.public_id
                    : "",
            url: posterUrl,
        },
        youtubeUrl,
        created_at:
            typeof data.created_at === "string"
                ? data.created_at
                : null,
        updated_at:
            typeof data.updated_at === "string"

export async function getSessions(): Promise<Session[]> {
    "use cache"
    const snapshot = await adminDb
        .collection("site_content")
        .doc("sessions")
        .get();

    if (!snapshot.exists) {
        return [];
    }

    const data = snapshot.data() as SessionsDocument;

    if (!Array.isArray(data.sessions)) {
        return [];
    }

    return data.sessions
        .map((item) => {
            if (!item || typeof item !== "object") {
                return null;
            }

            const session = item as SessionDocument;

            const id =
                typeof session.id === "string"
                    ? session.id
                    : "";

            if (!id) {
                return null;
            }

            return normalizeSession(id, session);
        })
        .filter(
            (session): session is Session =>
                session !== null
        )
        .sort((a, b) => {
            const dateA = a.created_at
                ? new Date(a.created_at).getTime()
                : 0;

            const dateB = b.created_at
                ? new Date(b.created_at).getTime()
                : 0;

            return dateB - dateA;
        });
}

    value: unknown,

    if (!value) {

        return value.toISOString();

        typeof value === "object" &&
        value !== null &&
        "toDate" in value &&
        typeof (

        if (data.is_active === false) {
            return;
        }

        sectorsById.set(
            snapshot.id,
            normalizeSector(
                snapshot.id,
                data,
            ),
        );
    });

    return sectorIds
        .map((id) =>
            sectorsById.get(id),
        )
        .filter(
            (
                sector,
            ): sector is ExhibitionSector =>
                Boolean(sector),
        );
}

                ),

                    introduction.description

import { adminDb } from "@/lib/firebase/admin";
import type { ServicesData } from "@/types/services/services";

function serializeFirestoreData(value: unknown): unknown {
    if (value === null || value === undefined) {
        return value;
    }

    if (value instanceof Date) {
        return value.toISOString();
    }

    if (Array.isArray(value)) {
        return value.map(serializeFirestoreData);
    }

    if (typeof value !== "object") {
        return value;
    }

    const objectValue = value as Record<string, unknown>;

    // Firestore Timestamp instance
    if (typeof objectValue.toDate === "function") {
        return (objectValue.toDate as () => Date)().toISOString();
    }

    // Firestore Timestamp represented by its internal fields
    const seconds =
        typeof objectValue._seconds === "number"
            ? objectValue._seconds
            : objectValue.seconds;

    const nanoseconds =
        typeof objectValue._nanoseconds === "number"
            ? objectValue._nanoseconds
            : objectValue.nanoseconds;

    if (
        typeof seconds === "number" &&
        typeof nanoseconds === "number"
    ) {
        return new Date(
            seconds * 1000 + nanoseconds / 1_000_000,
        ).toISOString();
    }

    // Convert nested objects into regular plain objects
    return Object.fromEntries(
        Object.entries(objectValue).map(([key, nestedValue]) => [
            key,
            serializeFirestoreData(nestedValue),
        ]),
    );
}

export async function getServicesData(): Promise<ServicesData | null> {
    "use cache";

    const snapshot = await adminDb
        .collection("site_content")
        .doc("services")
        .get();

    if (!snapshot.exists) {
        return null;
    }

    const data = snapshot.data();

    if (!data) {
        return null;
    }

    return serializeFirestoreData(data) as ServicesData;
}

import "server-only";

import { adminDb } from "@/lib/firebase/admin";
import type { HomepageSettings } from "@/lib/data/homepage";
import type { ThemeSettings } from "@/types/theme";

const themeFields: (keyof ThemeSettings)[] = [
    "arabicFont",
    "englishFont",
    "backgroundColor",
    "primaryColor",
    "secondaryColor",
    "textColor",
];

type ThemeDocumentData = {
    company: ThemeSettings;
    exhibitions: Record<string, Partial<ThemeSettings>>;
};

/* Validate Plain Object */
function isThemeObject(
    value: unknown
): value is Record<string, unknown> {
    return (
        typeof value === "object" &&
        value !== null &&
        !Array.isArray(value)
    );
}

/* Normalize Complete Theme */
function normalizeTheme(value: unknown): ThemeSettings {
    if (!isThemeObject(value)) {
        throw new Error("Invalid theme data received from Firestore.");
    }

    const theme = {} as ThemeSettings;

    for (const field of themeFields) {
        const fieldValue = value[field];

        if (
            typeof fieldValue !== "string" ||
            !fieldValue.trim()
        ) {
            throw new Error(
                `Theme is missing required field: ${field}`
            );
        }

        theme[field] = fieldValue;
    }

    return theme;
}

/* Normalize Partial Theme */
function normalizePartialTheme(
    value: unknown
): Partial<ThemeSettings> {
    if (!isThemeObject(value)) {
        return {};
    }

    const theme: Partial<ThemeSettings> = {};

    for (const field of themeFields) {
        const fieldValue = value[field];

        if (
            typeof fieldValue === "string" &&
            fieldValue.trim()
        ) {
            theme[field] = fieldValue;
        }
    }

    return theme;
}

/* Get Theme Document */
async function getThemeDocument(): Promise<ThemeDocumentData> {
    "use cache";

    const snapshot = await adminDb
        .collection("site_settings")
        .doc("settings_themes")
        .get();

    if (!snapshot.exists) {
        throw new Error("Theme settings document not found.");
    }

    const data = snapshot.data();

    if (!data) {
        throw new Error("Theme settings data is empty.");
    }

    const company = normalizeTheme(data.company);

    const rawExhibitions = isThemeObject(data.exhibitions)
        ? data.exhibitions
        : {};

    const exhibitions: Record<string, Partial<ThemeSettings>> = {};

    for (const [id, value] of Object.entries(rawExhibitions)) {
        exhibitions[id] = normalizePartialTheme(value);
    }

    // Return only normalized plain objects.
    // Firestore metadata such as updated_at is intentionally excluded.
    return {
        company,
        exhibitions,
    };
}

/* Get Company Theme */
function getCompanyTheme(
    data: ThemeDocumentData
): ThemeSettings {
    return data.company;
}

/* Get Exhibition Theme */
function getExhibitionTheme(
    data: ThemeDocumentData,
    exhibitionId: string
): Partial<ThemeSettings> | null {
    if (!exhibitionId) return null;

    return data.exhibitions[exhibitionId] ?? null;
}

/* Get Active Theme */
export async function getActiveTheme(
    settings: HomepageSettings
): Promise<ThemeSettings> {
    const data = await getThemeDocument();

    const companyTheme = getCompanyTheme(data);

    const exhibitionTheme = getExhibitionTheme(
        data,
        settings.exhibition_id
    );

    if (settings.mode === "exhibition" && exhibitionTheme) {
        return {
            ...companyTheme,
            ...exhibitionTheme,
        };
    }

    return companyTheme;
}

// proxy.ts
import { NextResponse } from "next/server";
import type { NextRequest } from "next/server";

const locales = ["ar", "en"] as const;

export function proxy(request: NextRequest) {
    const { pathname } = request.nextUrl;

    // Ignore Next.js internals and public files
    if (
        pathname.startsWith("/_next") ||
        pathname.startsWith("/api") ||
        pathname.includes(".")
    ) {
        return NextResponse.next();
    }

    let locale = "ar";

    // English URLs
    if (pathname === "/en" || pathname.startsWith("/en/")) {
        locale = "en";
    }

    const requestHeaders = new Headers(request.headers);
    requestHeaders.set("x-locale", locale);

    // English already matches the [locale] route
    if (locale === "en") {
        return NextResponse.next({
            request: {
                headers: requestHeaders,
            },
        });
    }

    // Arabic is the default locale.
    // Rewrite the public Arabic URL internally to /ar/...
    const url = request.nextUrl.clone();

    if (pathname === "/") {
        url.pathname = "/ar";
    } else {
        url.pathname = `/ar${pathname}`;
    }

    return NextResponse.rewrite(url, {
        request: {
            headers: requestHeaders,
        },
    });
}

export const config = {
    matcher: ["/((?!_next/static|_next/image|favicon.ico).*)"],
};
